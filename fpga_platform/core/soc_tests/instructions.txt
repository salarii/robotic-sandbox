# 1. Build the simulator from inside core/soc_tests/
cd ~/robot/fpga_platform/core/soc_tests
verilator --cc --exe --trace --build \
  --CFLAGS "-O3 -g" --Wno-WIDTH -Wno-fatal \
  --top-module sim_soc_zephyr \
  -I../wish_bone/uart \
  --Mdir obj_dir_dbg \
  ../sim_soc_zephyr.v \
  ../wish_bone/simple_to_wishbone.v \
  ../wish_bone/wishbone_plic.v \
  ../wish_bone/clint.v \
  ../wish_bone/wb_ram.v \
  ../wish_bone/wb_dp_ram.v \
  ../wish_bone/watchdog.v \
  ../wish_bone/uart/uart_top.v \
  ../wish_bone/uart/uart_wb.v \
  ../wish_bone/uart/uart_debug_if.v \
  ../wish_bone/uart/uart_receiver.v \
  ../wish_bone/uart/uart_regs.v \
  ../wish_bone/uart/uart_rfifo.v \
  ../wish_bone/uart/uart_sync_flops.v \
  ../wish_bone/uart/uart_tfifo.v \
  ../wish_bone/uart/uart_transmitter.v \
  ../wish_bone/uart/raminfr.v \
  ../cpus/VexRiscv.v \
  sim_soc_zephyr_dbg.cpp \
  -o sim_dbg

# 2. Run indefinitely (Ctrl+C to stop). 
# Note the ../../ to reach fpga_platform/build/zephyr/ from core/soc_tests/
./obj_dir/sim_soc_zephyr +hex=../../build/zephyr/hello/image.hex

# OR 3. Run for a finite number of cycles
SIM_TIMEOUT_CYCLES=50000 ./obj_dir/sim_soc_zephyr +hex=../../build/zephyr/hello/image.hex
